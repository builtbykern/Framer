interface MotionSpringValues {
    stiffness: number
    damping: number
    mass: number
}

/** NORMA_03 / kern-framer-motion-definitive presets (explicit values, no generic labels). */
const MOTION_SPRING_PRESETS: Record<
    Kern_QuoteIntakeProps["motionConfig"]["moreOptions"]["personality"],
    MotionSpringValues
> = {
    calm: { stiffness: 60, damping: 20, mass: 1.2 },
    balanced: { stiffness: 90, damping: 18, mass: 1.1 },
    snappy: { stiffness: 120, damping: 14, mass: 0.9 },
}

const MOTION_SPRING_INSTANT: MotionSpringValues = {
    stiffness: 10000,
    damping: 1000,
    mass: 0.01,
}

function resolveMotionSpring(
    personality: Kern_QuoteIntakeProps["motionConfig"]["moreOptions"]["personality"],
    overrides?: Partial<MotionSpringValues>
): MotionSpringValues {
    const base =
        MOTION_SPRING_PRESETS[personality] ?? MOTION_SPRING_PRESETS.balanced
    return {
        stiffness: overrides?.stiffness ?? base.stiffness,
        damping: overrides?.damping ?? base.damping,
        mass: overrides?.mass ?? base.mass,
    }
}

function clampLayoutDimension(
    value: number,
    floor: number,
    ceiling: number
): number {
    if (!Number.isFinite(value)) return floor
    return Math.min(ceiling, Math.max(floor, value))
}

interface QuoteContentColumnBounds {
    width: string
    minWidth: number
    maxWidth: number | undefined
}

function resolveQuoteContentColumnBounds(params: {
    showRail: boolean
    contentMinWidth: number
    contentMaxWidth: number
    globalMaxWidth: number
}): QuoteContentColumnBounds {
    const shellMax = clampLayoutDimension(params.globalMaxWidth, 400, 1600)
    const minW = clampLayoutDimension(params.contentMinWidth, 280, shellMax)
    const maxW = clampLayoutDimension(params.contentMaxWidth, minW, shellMax)

    if (params.showRail) {
        return { width: "100%", minWidth: 0, maxWidth: undefined }
    }

    return { width: "100%", minWidth: minW, maxWidth: maxW }
}

function resolveQuotePageMinHeight(minPageHeight: number): number | undefined {
    if (typeof minPageHeight === "number" && minPageHeight > 0) {
        return clampLayoutDimension(minPageHeight, 320, 1200)
    }
    return undefined
}

type BreakpointTier = "mobile" | "tablet" | "desktop"

function resolveBreakpointTier(
    containerWidth: number,
    mobileBreakpoint: number,
    tabletBreakpoint: number
): BreakpointTier {
    if (containerWidth < mobileBreakpoint) return "mobile"
    if (containerWidth < tabletBreakpoint) return "tablet"
    return "desktop"
}

function resolveTierNumber(
    containerWidth: number,
    mobileBreakpoint: number,
    tabletBreakpoint: number,
    desktop: number,
    tablet: number,
    mobile: number
): number {
    const tier = resolveBreakpointTier(
        containerWidth,
        mobileBreakpoint,
        tabletBreakpoint
    )
    if (tier === "mobile") return mobile
    if (tier === "tablet") return tablet
    return desktop
}

function resolveFluidFontSize(
    fontSizeDesktop: number,
    fontSizeTablet: number,
    fontSizeMobile: number,
    containerWidth: number,
    mobileBreakpoint: number,
    tabletBreakpoint: number
): string {
    const desktop = clampLayoutDimension(fontSizeDesktop, 10, 96)
    const tablet = clampLayoutDimension(fontSizeTablet, 10, 96)
    const mobile = clampLayoutDimension(fontSizeMobile, 10, 96)
    const minSize = Math.min(mobile, tablet, desktop)
    const maxSize = Math.max(mobile, tablet, desktop)
    const mid = Math.round((tablet + desktop) / 2)
    return `clamp(${minSize}px, calc(${mid}px + (100cqi - ${tabletBreakpoint}px) * 0.015), ${maxSize}px)`
}

interface QuoteEstimateEngineResult {
    baseSubtotal: number
    totalEstimate: number
    lineContributions: LineContribution[]
    intentMult: number
    timelineMult: number
    rushMult: number
}

function computeQuoteEstimateEngine(
    formData: FormDataState,
    formulaLineItems: SimpleFormulaLineItem[],
    showAdvancedPricing: boolean,
    intentMultiplierOverrides: KeyMultiplierRow[],
    timelineMultiplierOverrides: KeyMultiplierRow[],
    rushMultiplierWhenChecked: number
): QuoteEstimateEngineResult {
    const lineContributionsBase = formulaLineItems.map((lineItem) => {
        const raw = formData[lineItem.quantityFieldKey]
        const quantity =
            typeof raw === "number"
                ? raw
                : typeof raw === "string"
                  ? Number.parseFloat(raw) || 0
                  : 0
        const lineSubtotal = quantity * lineItem.unitPrice
        return {
            label: lineItem.label,
            quantityFieldKey: lineItem.quantityFieldKey,
            quantity,
            unitPrice: lineItem.unitPrice,
            lineSubtotal,
            contributionAfterMultipliers: 0,
        }
    })

    const baseSubtotal = lineContributionsBase.reduce(
        (sum, line) => sum + line.lineSubtotal,
        0
    )

    const intentKey = String(formData.intent ?? "")
    const timelineKey = String(formData.timeline ?? "")

    const lookupMult = (rows: KeyMultiplierRow[], key: string): number => {
        const hit = rows.find((row) => row.key === key)
        if (!hit || typeof hit.multiplier !== "number") return 1
        return Number.isFinite(hit.multiplier) && hit.multiplier > 0
            ? hit.multiplier
            : 1
    }

    const intentMult = showAdvancedPricing
        ? lookupMult(intentMultiplierOverrides, intentKey)
        : 1
    const timelineMult = showAdvancedPricing
        ? lookupMult(timelineMultiplierOverrides, timelineKey)
        : 1

    const rushBase =
        typeof rushMultiplierWhenChecked === "number" &&
        Number.isFinite(rushMultiplierWhenChecked) &&
        rushMultiplierWhenChecked > 0
            ? rushMultiplierWhenChecked
            : 1.3
    const rushMult = formData.rush === true ? rushBase : 1

    const combined = baseSubtotal * intentMult * timelineMult * rushMult
    const totalEstimate = Math.round(combined)

    let allocated = 0
    const n = lineContributionsBase.length
    const lineContributions: LineContribution[] =
        baseSubtotal <= 0 || n === 0
            ? lineContributionsBase.map((line) => ({
                  ...line,
                  contributionAfterMultipliers: 0,
              }))
            : lineContributionsBase.map((line, index) => {
                  const isLast = index === n - 1
                  const share = line.lineSubtotal / baseSubtotal
                  const portion = isLast
                      ? Math.max(0, totalEstimate - allocated)
                      : Math.round(totalEstimate * share)
                  if (!isLast) allocated += portion
                  return {
                      ...line,
                      contributionAfterMultipliers: Math.max(0, portion),
                  }
              })

    return {
        baseSubtotal,
        totalEstimate,
        lineContributions,
        intentMult,
        timelineMult,
        rushMult,
    }
}

function validateEmailAddress(email: string): boolean {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
}

function formatEstimateRange(
    totalEstimate: number,
    rangeLowPercent: number,
    rangeHighPercent: number,
    currencySymbol: string
): { rangeLow: number; rangeHigh: number; text: string } {
    const rangeLow = Math.round(totalEstimate * rangeLowPercent)
    const rangeHigh = Math.round(totalEstimate * rangeHighPercent)
    const text = `${currencySymbol}${rangeLow.toLocaleString()} – ${currencySymbol}${rangeHigh.toLocaleString()}`
    return { rangeLow, rangeHigh, text }
}

function validateQuoteStep(
    step: number,
    formData: FormDataState,
    quantityFields: QuantityField[],
    showContactFields: boolean
): Record<string, string | undefined> {
    const errors: Record<string, string | undefined> = {}

    if (step === 1) {
        if (!formData.intent) {
            errors.intent = "Select a project type to continue"
        }
    } else if (step === 2) {
        for (const field of quantityFields) {
            const value = formData[field.key]
            if (value === undefined || value === null || value === "") {
                errors[field.key] = "Enter a value for this field"
            } else {
                const n =
                    typeof value === "number"
                        ? value
                        : Number.parseFloat(String(value))
                if (Number.isNaN(n)) {
                    errors[field.key] = "Enter a valid number"
                } else if (n < field.min) {
                    errors[field.key] = `Minimum value is ${field.min}`
                } else if (n > field.max) {
                    errors[field.key] = `Maximum value is ${field.max}`
                }
            }
        }
    } else if (step === 3) {
        if (!formData.timeline) {
            errors.timeline = "Select a delivery window"
        }
    } else if (step === 4 && showContactFields) {
        if (!formData.email || String(formData.email).trim() === "") {
            errors.email = "Enter your email address"
        } else if (!validateEmailAddress(String(formData.email))) {
            errors.email = "Enter a valid email address"
        }
        if (!formData.consent) {
            errors.consent = "Confirm before continuing"
        }
    }

    return errors
}

interface QuoteOptionTileProps {
    optionKey: string

    helperText?: string
    selected: boolean
    onSelect: () => void
    allowMicroMotion: boolean
    colors: {
        textPrimary: string
        textSecondary: string
        moreOptions?: {
            focusRingColor?: string
        }
    }
    cardFont: CSSProperties
    borderRadius: number
    windowWidth: number
}

function QuoteOptionTile(props: QuoteOptionTileProps) {
    const {
        optionKey,
        label,
        helperText,
        selected,
        onSelect,
        allowMicroMotion,
        colors,
        cardFont,
        borderRadius,
        windowWidth,
    } = props

    const focusRingColor = colors.moreOptions?.focusRingColor ?? "#C9A227"

    return (
        <motion.button
            type="button"
            role="radio"
            aria-checked={selected}
            whileHover={allowMicroMotion ? { y: -1 } : undefined}
            whileTap={allowMicroMotion ? { scale: 0.99 } : undefined}
            transition={allowMicroMotion ? { duration: 0.15 } : undefined}
            onClick={onSelect}
            style={{
                ...cardFont,
                padding: windowWidth < 480 ? 14 : 16,
                borderRadius,
                border: `${selected ? OPTION_TILE_STYLES.selected.borderWidth : OPTION_TILE_STYLES.idle.borderWidth} solid ${selected ? OPTION_TILE_STYLES.selected.borderColor : OPTION_TILE_STYLES.idle.borderColor}`,
                backgroundColor: selected
                    ? OPTION_TILE_STYLES.selected.backgroundColor
                    : OPTION_TILE_STYLES.idle.backgroundColor,
                boxShadow: selected
                    ? OPTION_TILE_STYLES.selected.boxShadow
                    : "none",
                color: colors.textPrimary,
                cursor: "pointer",
                textAlign: "left",
                display: "flex",
                flexDirection: "column",
                gap: 8,
                outline: `2px solid transparent`,
                outlineOffset: 2,
                fontWeight: selected
                    ? Number(cardFont.fontWeight || 600) * 1.05
                    : cardFont.fontWeight,
            }}
            onFocus={(e) => {
                e.currentTarget.style.outline = `2px solid ${focusRingColor}`
            }}
            onBlur={(e) => {
                e.currentTarget.style.outline = "2px solid transparent"
            }}
            onMouseEnter={(e) => {
                if (allowMicroMotion && !selected) {
                    e.currentTarget.style.borderColor =
                        OPTION_TILE_STYLES.hover.borderColor
                }
            }}
            onMouseLeave={(e) => {
                if (!selected) {
                    e.currentTarget.style.borderColor =
                        OPTION_TILE_STYLES.idle.borderColor
                }
            }}
        >
            <div>{label}</div>
            {helperText ? (
                <div
                    style={{
                        fontSize: windowWidth < 480 ? "12px" : "12px",
                        fontWeight: 400,
                        color: colors.textSecondary,
                        lineHeight: "1.4em",
                    }}
                >
                    {helperText}
                </div>
            ) : null}
        </motion.button>
    )
}

interface QuoteAnimatedEstimateProps {
    totalEstimate: number
    rangeLowPercent: number
    rangeHighPercent: number
    currencySymbol: string
    isOnCanvas: boolean
    prefersReducedMotion: boolean
    springConfig: {
        stiffness: number
        damping: number
        mass: number
    }
    windowWidth: number
    textColor: string
    estimateFont: CSSProperties
    variant: "compact" | "hero"
}

function QuoteAnimatedEstimate(props: QuoteAnimatedEstimateProps) {
    const {
        totalEstimate,
        rangeLowPercent,
        rangeHighPercent,
        currencySymbol,
        isOnCanvas,
        prefersReducedMotion,
        springConfig,
        windowWidth,
        textColor,
        estimateFont,
        variant,
    } = props

    const totalTarget = useMotionValue(0)
    const totalSpring = useSpring(totalTarget, springConfig)

    const [smoothedTotal, setSmoothedTotal] = useState(totalEstimate)

    useLayoutEffect(() => {
        totalTarget.set(totalEstimate)
    }, [totalEstimate, totalTarget])

    useMotionValueEvent(totalSpring, "change", (v) => {
        if (!isOnCanvas && !prefersReducedMotion) {
            startTransition(() => {
                setSmoothedTotal(Math.round(v))
            })
        }
    })

    useLayoutEffect(() => {
        if (isOnCanvas || prefersReducedMotion) {
            startTransition(() => {
                setSmoothedTotal(totalEstimate)
            })
        }
    }, [isOnCanvas, prefersReducedMotion, totalEstimate])

    const displayTotal =
        isOnCanvas || prefersReducedMotion ? totalEstimate : smoothedTotal

    const { text: rangeText } = formatEstimateRange(
        displayTotal,
        rangeLowPercent,
        rangeHighPercent,
        currencySymbol
    )

    const isCompact = variant === "compact"

    return (
        <motion.span
            aria-live="polite"
            aria-atomic="true"
            style={{
                ...estimateFont,
                color: textColor,
                fontVariantNumeric: "tabular-nums",
                letterSpacing: "-0.02em",
                lineHeight: "1.15em",
                display: "block",
                willChange:
                    isOnCanvas || prefersReducedMotion
                        ? "auto"
                        : "transform, opacity",
                ...(isCompact
                    ? {
                          fontSize: windowWidth < 480 ? "13px" : "15px",
                          fontWeight: 600,
                          whiteSpace: "nowrap",
                          overflow: "hidden",
                          textOverflow: "ellipsis",
                          maxWidth: "100%",
                          textAlign: "right",
                          marginTop: 0,
                      }
                    : {
                          fontSize:
                              windowWidth < 480
                                  ? "22px"
                                  : windowWidth < 720
                                    ? "26px"
                                    : "30px",
                          fontWeight: 600,
                          marginTop: 4,
                          wordBreak: "break-word",
                      }),
            }}
        >
            {rangeText}
        </motion.span>
    )
}

            resolveFluidFontSize(
                typography.labelFontSizeDesktop,
                typography.labelFontSizeTablet,
                typography.labelFontSizeMobile,
                containerWidth,
                mobileBreakpoint,
                tabletBreakpoint
            ),
        [
            typography.labelFontSizeDesktop,
            typography.labelFontSizeTablet,
            typography.labelFontSizeMobile,
            containerWidth,
            mobileBreakpoint,
            tabletBreakpoint,
        ]
    )

    const contentColumnBounds = useMemo(
        () =>
            resolveQuoteContentColumnBounds({
                showRail,
                contentMinWidth,
                contentMaxWidth,
                globalMaxWidth,
            }),
        [showRail, contentMinWidth, contentMaxWidth, globalMaxWidth]
    )

    const pageMinHeight = useMemo(
        () => resolveQuotePageMinHeight(minPageHeight),
        [minPageHeight]
    )

    const {
        pagePadding: responsivePadding,
        cardPadding: responsiveCardPadding,
        sectionGap: responsiveSectionGap,
        fieldGap: responsiveFieldGap,
        buttonHeight: responsiveButtonHeight,
    } = layoutSpacing

    const backdropBlur = useMemo(() => {
        if (!cardStyle.showBackground) return 0
        if (prefersReducedMotion) return 0
        if (isOnCanvas) return Math.min(cardStyle.backdropBlur, 16)
        return cardStyle.backdropBlur
    }, [
        isOnCanvas,
        prefersReducedMotion,
        cardStyle.showBackground,
        cardStyle.backdropBlur,
    ])

    const boxShadow = useMemo(() => {
        if (!cardStyle.showShadow) return "none"
        const intensity = cardStyle.shadowIntensity
        return `0 ${8 * intensity}px ${32 * intensity}px rgba(0, 0, 0, ${0.2 * intensity}), 0 ${2 * intensity}px ${8 * intensity}px rgba(0, 0, 0, ${0.15 * intensity})`
    }, [cardStyle.showShadow, cardStyle.shadowIntensity])

    const estimateEngine = useMemo(
        () =>
            computeQuoteEstimateEngine(
                formData,
                formulaLineItems,
                showAdvancedPricing,
                intentMultiplierOverrides,
                timelineMultiplierOverrides,
                rushMultiplierWhenChecked
            ),
        [
            formData,
            formulaLineItems,
            showAdvancedPricing,
            intentMultiplierOverrides,
            timelineMultiplierOverrides,
            rushMultiplierWhenChecked,
        ]
    )

    const totalEstimate = estimateEngine.totalEstimate

    const rangeLowPercent = useMemo(() => {
        const v = estimateDisplay.rangeLowPercent
        return typeof v === "number" && Number.isFinite(v) && v > 0 ? v : 0.9
    }, [estimateDisplay.rangeLowPercent])

    const rangeHighPercent = useMemo(() => {
        const v = estimateDisplay.rangeHighPercent
        return typeof v === "number" && Number.isFinite(v) && v > 0 ? v : 1.1
    }, [estimateDisplay.rangeHighPercent])

    const breakdownStaggerSec = useMemo(() => {
        const ms = estimateDisplay.breakdownStaggerMs
        if (typeof ms === "number" && Number.isFinite(ms) && ms >= 0) {
            return ms / 1000
        }
        return 0.04
    }, [estimateDisplay.breakdownStaggerMs])

    const currencySymbol =
        estimateDisplay.currencySymbol &&
        estimateDisplay.currencySymbol.trim() !== ""
            ? estimateDisplay.currencySymbol
            : "$"

    const motionSpring = useMemo(
        () =>
            resolveMotionSpring(motionConfig.moreOptions.personality, {
                stiffness: motionConfig.moreOptions.stiffness,
                damping: motionConfig.moreOptions.damping,
                mass: motionConfig.moreOptions.mass,
            }),
        [motionConfig.moreOptions]
    )

    const estimateSpringConfig = useMemo(
        () =>
            isOnCanvas || prefersReducedMotion
                ? MOTION_SPRING_INSTANT
                : motionSpring,
        [isOnCanvas, prefersReducedMotion, motionSpring]
    )

    const allowMicroMotion = !isOnCanvas && !prefersReducedMotion

    const estimateFont = typography.labelFont

    const stepLabels = [
        narrative.step1Label,
        narrative.step2Label,
        narrative.step3Label,
        narrative.step4Label,
    ]

    const successSpring = useMemo(() => {
        if (isOnCanvas || prefersReducedMotion) {
            return { duration: 0 }
        }
        const spring = MOTION_SPRING_PRESETS.calm
        return {
            type: "spring" as const,
            stiffness: spring.stiffness,
            damping: spring.damping,
            mass: spring.mass,
        }
    }, [isOnCanvas, prefersReducedMotion])

    const stepTransition = useMemo(() => {
        if (isOnCanvas || prefersReducedMotion) {
            return {
                initial: { opacity: 1, y: 0 },
                animate: { opacity: 1, y: 0 },
                exit: { opacity: 1, y: 0 },
                transition: { duration: 0 },
            }
        }
        const intensity =
            typeof motionConfig.intensity === "number" &&
            Number.isFinite(motionConfig.intensity)
                ? Math.max(0.25, Math.min(motionConfig.intensity, 2))
                : 1
        const yEnter = Math.min(12, 12 * intensity)
        const yExit = Math.min(12, 8 * intensity)
        return {
            initial: { opacity: 0, y: yEnter },
            animate: { opacity: 1, y: 0 },
            exit: { opacity: 0, y: -yExit },
            transition: {
                type: "spring" as const,
                stiffness: motionSpring.stiffness,
                damping: motionSpring.damping,
                mass: motionSpring.mass,
            },
        }
    }, [isOnCanvas, prefersReducedMotion, motionConfig.intensity, motionSpring])

    const validateCurrentStep = useCallback((): boolean => {
        const errors = validateQuoteStep(
            currentStep,
            formData,
            quantityFields,
            showContactFields
        )
        startTransition(() => setValidationErrors(errors))
        return Object.keys(errors).length === 0
    }, [currentStep, formData, quantityFields, showContactFields])

    const handleNext = useCallback(() => {
        startTransition(() => setAttemptedNext(true))
        if (!validateCurrentStep()) return

        if (currentStep < 4) {
            startTransition(() => {
                setCurrentStep(currentStep + 1)
                setAttemptedNext(false)
                setValidationErrors({})
            })
            return
        }

        startTransition(() => {
            setIsSubmitted(true)
            setAttemptedNext(false)
            setValidationErrors({})
        })
    }, [currentStep, validateCurrentStep])

    const handleBack = useCallback(() => {
        if (currentStep > 1) {
            startTransition(() => {
                setCurrentStep(currentStep - 1)
                setAttemptedNext(false)
                setValidationErrors({})
            })
        }
    }, [currentStep])

    const onPatchFormField = useCallback((key: string, value: FormValue) => {
        startTransition(() => {
            setFormData((prev) => ({
                ...prev,
                [key]: value,
            }))
        })
    }, [])

    const onClearFieldError = useCallback((key: string) => {
        startTransition(() => {
            setValidationErrors((prev) => ({
                ...prev,
                [key]: undefined,
            }))
        })
    }, [])

    const stepScreens = useMemo(
        () => [step1, step2, step3, step4],
        [step1, step2, step3, step4]
    )
    const activeStepScreen = stepScreens[currentStep - 1] ?? step4
    const currentScreenTitle = activeStepScreen.screenTitle
    const currentScreenSubtitle = activeStepScreen.screenSubtitle

    const visuallyHidden: CSSProperties = {
        position: "absolute",
        width: 1,
        height: 1,
        padding: 0,
        margin: -1,
        overflow: "hidden",
        clipPath: "inset(50%)",
        whiteSpace: "nowrap",
        border: 0,
    }

    const formattedEstimate = formatEstimateRange(
        totalEstimate,
        rangeLowPercent,
        rangeHighPercent,
        currencySymbol
    )

    const showStepIndexInCard = !(showRail && narrative.showStepList)

    const stepLiveMessage =
        announceStepChanges && !isSubmitted
            ? `Step ${currentStep} of 4: ${currentScreenTitle}`
            : ""
    const estimateLiveMessage =
        announceEstimateRange && currentStep === 4 && !isSubmitted
            ? `Indicative range ${formattedEstimate.text}`
            : ""

    const secondaryBg =
        colors.moreOptions?.secondaryButtonColor ?? "rgba(247, 244, 237, 0.08)"
    const secondaryText =
        colors.moreOptions?.secondaryButtonTextColor ?? colors.textPrimary

    const pageGlowIntensity = useMemo(() => {
        const intensity = cardStyle.moreOptions?.pageGlowIntensity ?? 1
        return Math.max(0, Math.min(intensity, 2))
    }, [cardStyle.moreOptions?.pageGlowIntensity])

    const railStyle: CSSProperties = {
        width: railWidth,
        flexShrink: 0,
        display: "flex",
        flexDirection: "column",
        gap: responsiveSectionGap * 1.5,
        paddingRight: railGap,
        borderRight: `1px solid ${colors.border}`,
    }

    const renderProgressBlock = (compact: boolean) => (
        <div
            style={{
                display: "flex",
                flexDirection: "column",
                gap: responsiveFieldGap,
                marginBottom: compact ? 0 : responsiveSectionGap,
            }}
        >
            {compact ? (
                <div
                    style={{
                        ...typography.smallPrintFont,
                        color: colors.textSecondary,
                        textTransform: "uppercase",
                        fontVariantNumeric: "tabular-nums",
                        letterSpacing: "0.05em",
                        fontSize: containerWidth < 480 ? "10px" : "11px",
                        opacity: 0.7,
                    }}
                    aria-hidden="true"
                >
                    {String(currentStep).padStart(2, "0")} / 04
                </div>
            ) : null}
            <div
                role="progressbar"
                aria-label="Form progress"
                aria-valuemin={1}
                aria-valuemax={4}
                aria-valuenow={currentStep}
                style={{ display: "flex", gap: 4 }}
            >
                {stepLabels.map((_, index) => {
                    const stepNumber = index + 1
                    const isComplete = stepNumber < currentStep
                    const isActive = stepNumber === currentStep
                    return (
                        <div
                            key={stepNumber}
                            style={{
                                flex: 1,
                                height: 3,
                                borderRadius: 1.5,
                                backgroundColor:
                                    isComplete || isActive
                                        ? colors.accent
                                        : colors.border,
                                opacity: isComplete || isActive ? 1 : 0.15,
                                transition:
                                    isOnCanvas || prefersReducedMotion
                                        ? "none"
                                        : "opacity 0.25s ease, background-color 0.25s ease",
                            }}
                        />
                    )
                })}
            </div>
            {narrative.showStepList && (
                <div
                    style={{
                        display: "flex",
                        flexDirection: "column",
                        gap: compact ? 8 : 10,
                    }}
                >
                    {stepLabels.map((label, index) => {
                        const stepNumber = index + 1
                        const isActive = stepNumber === currentStep
                        const isComplete = stepNumber < currentStep
                        return (
                            <div
                                key={label}
                                style={{
                                    display: "flex",
                                    alignItems: "center",
                                    gap: 10,
                                    opacity: isActive
                                        ? 1
                                        : isComplete
                                          ? 0.55
                                          : 0.35,
                                }}
                            >
                                <div
                                    style={{
                                        width: 22,
                                        height: 22,
                                        borderRadius: "50%",
                                        border: `1px solid ${isActive ? colors.accent : colors.border}`,
                                        color: isActive
                                            ? colors.accent
                                            : colors.textSecondary,
                                        fontSize: 11,
                                        display: "flex",
                                        alignItems: "center",
                                        justifyContent: "center",
                                        ...typography.smallPrintFont,
                                    }}
                                >
                                    {stepNumber}
                                </div>
                                <span
                                    style={{
                                        ...typography.labelFont,
                                        color: isActive
                                            ? colors.textPrimary
                                            : colors.textSecondary,
                                        fontWeight: isActive ? 600 : 400,
                                    }}
                                >
                                    {label}
                                </span>
                            </div>
                        )
                    })}
                </div>
            )}
        </div>
    )

    const renderStepBody = () => (
        <AnimatePresence mode="wait">
            <motion.div
                ref={stepBodyRef}
                key={currentStep}
                initial={
                    isOnCanvas || prefersReducedMotion || !isStepBodyInView
                        ? { opacity: 1, y: 0 }
                        : stepTransition.initial
                }
                animate={stepTransition.animate}
                exit={stepTransition.exit}
                transition={stepTransition.transition}
                style={{
                    display: "flex",
                    flexDirection: "column",
                    gap: responsiveSectionGap,
                    willChange:
                        isOnCanvas || prefersReducedMotion
                            ? "auto"
                            : isStepBodyInView
                              ? "transform, opacity"
                              : "auto",
                }}
            >
                <div
                    style={{
                        display: "flex",
                        flexDirection: "column",
                        gap: responsiveFieldGap * 0.5,
                    }}
                >
                    {showStepIndexInCard ? (
                        <div
                            style={{
                                ...typography.smallPrintFont,
                                color: colors.textSecondary,
                                textTransform: "uppercase",
                                fontVariantNumeric: "tabular-nums",
                                letterSpacing: "0.05em",
                                fontSize:
                                    containerWidth < 480 ? "10px" : "11px",
                                opacity: 0.7,
                            }}
                            aria-hidden="true"
                        >
                            {String(currentStep).padStart(2, "0")} / 04
                        </div>
                    ) : null}
                    <div
                        ref={stepHeadingRef}
                        tabIndex={-1}
                        style={{
                            ...typography.titleFont,
                            color: colors.textPrimary,
                            fontSize: responsiveTitleFontSize,
                            outline: "none",
                        }}
                        id={`step-${currentStep}-title`}
                    >
                        {currentScreenTitle}
                    </div>
                    <div
                        style={{
                            ...typography.labelFont,
                            color: colors.textSecondary,
                            fontSize: responsiveLabelFontSize,
                        }}
                    >
                        {currentScreenSubtitle}
                    </div>
                </div>

                {currentStep === 4 && !showContactFields && disclaimerText ? (
                    <div
                        style={{
                            ...typography.smallPrintFont,
                            color: colors.textSecondary,
                            opacity: 0.75,
                            maxWidth: "40em",
                            lineHeight: "1.5em",
                            fontSize: containerWidth < 480 ? "12px" : undefined,
                        }}
                    >
                        {disclaimerText}
                    </div>
                ) : null}

                {(currentStep === 2 || currentStep === 3) && (
                    <div
                        style={{
                            display: "flex",
                            flexDirection:
                                containerWidth < 600 ? "column" : "row",
                            justifyContent: "space-between",
                            alignItems:
                                containerWidth < 600 ? "flex-start" : "center",
                            gap: responsiveFieldGap,
                            paddingTop: responsiveFieldGap * 0.75,
                            paddingBottom: responsiveFieldGap * 0.75,
                            borderTop: `1px solid ${colors.border}`,
                            borderBottom: `1px solid ${colors.border}`,
                            opacity: 0.85,
                        }}
                    >
                        <div
                            style={{
                                ...typography.smallPrintFont,
                                color: colors.textSecondary,
                                textTransform: "uppercase",
                                letterSpacing: "0.05em",
                                fontSize:
                                    containerWidth < 480 ? "10px" : "11px",
                            }}
                        >
                            Indicative range
                        </div>
                        <div
                            style={{
                                flex: 1,
                                minWidth: 0,
                                display: "flex",
                                justifyContent: "flex-end",
                            }}
                        >
                            <QuoteAnimatedEstimate
                                totalEstimate={totalEstimate}
                                rangeLowPercent={rangeLowPercent}
                                rangeHighPercent={rangeHighPercent}
                                currencySymbol={currencySymbol}
                                isOnCanvas={isOnCanvas}
                                prefersReducedMotion={prefersReducedMotion}
                                springConfig={estimateSpringConfig}
                                windowWidth={containerWidth}
                                textColor={colors.textPrimary}
                                estimateFont={estimateFont}
                                variant="compact"
                            />
                        </div>
                    </div>
                )}

                {currentStep === 1 && (
                    <QuoteStep1Intent
                        step1={step1}
                        intentOptions={intentOptions}
                        stepTitleId={`step-${currentStep}-title`}
                        colors={colors}
                        typography={typography}
                        responsiveSectionGap={responsiveSectionGap}
                        responsiveFieldGap={responsiveFieldGap}
                        borderRadius={borderRadius}
                        responsiveButtonHeight={responsiveButtonHeight}
                        windowWidth={containerWidth}
                        isOnCanvas={isOnCanvas}
                        prefersReducedMotion={prefersReducedMotion}
                        allowMicroMotion={allowMicroMotion}
                        formData={formData}
                        validationErrors={validationErrors}
                        attemptedNext={attemptedNext}
                        onPatchFormField={onPatchFormField}
                        onClearFieldError={onClearFieldError}
                    />
                )}

                {currentStep === 2 && (
                    <QuoteStep2Scope
                        quantityFields={quantityFields}
                        quantityControlType={quantityControlType}
                        colors={colors}
                        typography={typography}
                        responsiveSectionGap={responsiveSectionGap}
                        responsiveFieldGap={responsiveFieldGap}
                        borderRadius={borderRadius}
                        responsiveButtonHeight={responsiveButtonHeight}
                        windowWidth={containerWidth}
                        isOnCanvas={isOnCanvas}
                        prefersReducedMotion={prefersReducedMotion}
                        allowMicroMotion={allowMicroMotion}
                        formData={formData}
                        validationErrors={validationErrors}
                        attemptedNext={attemptedNext}
                        onPatchFormField={onPatchFormField}
                        onClearFieldError={onClearFieldError}
                    />
                )}

                {currentStep === 3 && (
                    <QuoteStep3Timeline
                        step3={step3}
                        timelineBucketOptions={timelineBucketOptions}
                        colors={colors}
                        typography={typography}
                        responsiveSectionGap={responsiveSectionGap}
                        responsiveFieldGap={responsiveFieldGap}
                        borderRadius={borderRadius}
                        responsiveButtonHeight={responsiveButtonHeight}
                        windowWidth={containerWidth}
                        isOnCanvas={isOnCanvas}
                        prefersReducedMotion={prefersReducedMotion}
                        allowMicroMotion={allowMicroMotion}
                        formData={formData}
                        validationErrors={validationErrors}
                        attemptedNext={attemptedNext}
                        onPatchFormField={onPatchFormField}
                        onClearFieldError={onClearFieldError}
                    />
                )}

                {currentStep === 4 && showContactFields ? (
                    <QuoteStep4Contact
                        step4={step4}
                        colors={colors}
                        typography={typography}
                        responsiveSectionGap={responsiveSectionGap}
                        responsiveFieldGap={responsiveFieldGap}
                        borderRadius={borderRadius}
                        responsiveButtonHeight={responsiveButtonHeight}
                        windowWidth={containerWidth}
                        isOnCanvas={isOnCanvas}
                        prefersReducedMotion={prefersReducedMotion}
                        allowMicroMotion={allowMicroMotion}
                        formData={formData}
                        validationErrors={validationErrors}
                        attemptedNext={attemptedNext}
                        onPatchFormField={onPatchFormField}
                        onClearFieldError={onClearFieldError}
                    />
                ) : null}

                {currentStep === 4 ? (
                    <QuoteStep4ReviewFooter
                        lineContributions={estimateEngine.lineContributions}
                        showLineBreakdown={estimateDisplay.showLineBreakdown}
                        breakdownStaggerSec={breakdownStaggerSec}
                        totalEstimate={totalEstimate}
                        rangeLowPercent={rangeLowPercent}
                        rangeHighPercent={rangeHighPercent}
                        currencySymbol={currencySymbol}
                        isOnCanvas={isOnCanvas}
                        prefersReducedMotion={prefersReducedMotion}
                        springConfig={estimateSpringConfig}
                        colors={{
                            textPrimary: colors.textPrimary,
                            textSecondary: colors.textSecondary,
                        }}
                        typography={{
                            smallPrintFont: typography.smallPrintFont,
                            labelFont: typography.labelFont,
                        }}
                        windowWidth={containerWidth}
                        responsiveSectionGap={responsiveSectionGap}
                        responsiveFieldGap={responsiveFieldGap}
                        intentOptions={intentOptions}
                        quantityFields={quantityFields}
                        formData={formData}
                    />
                ) : null}

                <QuoteStepNav
                    currentStep={currentStep}
                    backButtonLabel={backButtonLabel}
                    primaryButtonLabels={{
                        step1: step1.primaryButtonLabel,
                        step2: step2.primaryButtonLabel,
                        step3: step3.primaryButtonLabel,
                        step4: step4PrimaryLabel,
                    }}
                    onBack={handleBack}
                    onNext={handleNext}
                    colors={{
                        accent: colors.accent,
                        backgroundColor: colors.backgroundColor,
                        border: colors.border,
                        moreOptions: colors.moreOptions,
                    }}
                    typography={{ buttonFont: typography.buttonFont }}
                    responsiveButtonHeight={responsiveButtonHeight}
                    responsiveFieldGap={responsiveFieldGap}
                    borderRadius={borderRadius}
                />
            </motion.div>
        </AnimatePresence>
    )

    return (
        <div
            ref={rootRef}
            role="region"
            aria-label={ariaLabel || narrative.productTitle}
            style={{
                width: "100%",
                height: "auto",
                minHeight: pageMinHeight,
                maxHeight: "100%",
                position: "relative",
                display: "flex",
                flexDirection: "column",
                justifyContent: "flex-start",
                alignItems: "stretch",
                backgroundColor: colors.backgroundColor,
                padding: responsivePadding,
                overflow: "hidden",
                boxSizing: "border-box",
                ...style,
            }}
        >
            {/* Soft radial gradient background layer */}
            <div
                style={{
                    position: "absolute",
                    inset: 0,
                    pointerEvents: "none",
                    zIndex: 0,
                    opacity: isOnCanvas
                        ? 0.5 * pageGlowIntensity
                        : pageGlowIntensity,
                }}
            >
                <div
                    style={{
                        position: "absolute",
                        top: "-20%",
                        left: "-10%",
                        width: "50%",
                        height: "50%",
                        background: `radial-gradient(circle, ${colors.accent}12 0%, transparent 70%)`,
                        filter: "blur(40px)",
                    }}
                />
                <div
                    style={{
                        position: "absolute",
                        bottom: "-20%",
                        right: "-10%",
                        width: "50%",
                        height: "50%",
                        background: `radial-gradient(circle, ${colors.accent}0F 0%, transparent 70%)`,
                        filter: "blur(40px)",
                    }}
                />
            </div>

            <div
                style={{
                    width: "100%",
                    maxWidth: globalMaxWidth,
                    margin: "0 auto",
                    display: "flex",
                    flexDirection: showRail ? "row" : "column",
                    alignItems: showRail ? "stretch" : "center",
                    gap: showRail ? 0 : responsiveSectionGap,
                    position: "relative",
                    zIndex: 1,
                    flex: "0 0 auto",
                }}
            >
                {showRail ? (
                    <aside style={railStyle}>
                        <div
                            style={{
                                ...typography.titleFont,
                                color: colors.textPrimary,
                                fontSize: responsiveTitleFontSize,
                                lineHeight: "1.08em",
                            }}
                        >
                            {narrative.productTitle}
                        </div>
                        <div
                            style={{
                                ...typography.labelFont,
                                color: colors.textSecondary,
                                fontSize: "13px",
                                opacity: 0.62,
                            }}
                        >
                            {narrative.supportingLine}
                        </div>
                        {narrative.showStepList
                            ? renderProgressBlock(true)
                            : null}
                    </aside>
                ) : null}
                <div
                    style={{
                        flex: 1,
                        minWidth: contentColumnBounds.minWidth,
                        width: contentColumnBounds.width,
                        maxWidth: contentColumnBounds.maxWidth,
                    }}
                >
                    {stepLiveMessage ? (
                        <div
                            aria-live="polite"
                            aria-atomic="true"
                            style={visuallyHidden}
                        >
                            {stepLiveMessage}
                        </div>
                    ) : null}
                    {estimateLiveMessage ? (
                        <div
                            aria-live="polite"
                            aria-atomic="true"
                            style={visuallyHidden}
                        >
                            {estimateLiveMessage}
                        </div>
                    ) : null}
                    <QuoteCardFrame
                        showBackground={cardStyle.showBackground}
                        backdropBlur={backdropBlur}
                        showShadow={cardStyle.showShadow}
                        shadowIntensity={cardStyle.shadowIntensity}
                        cardBackground={colors.cardBackground}
                        border={colors.border}
                        borderRadius={borderRadius}
                        responsiveCardPadding={responsiveCardPadding}
                    >
                        {previewImageUrl.trim().length > 0 ? (
                            <img
                                src={previewImageUrl}
                                alt={previewImageAlt}
                                style={{
                                    width: "100%",
                                    maxHeight: 120,
                                    objectFit: "cover",
                                    borderRadius: borderRadius * 0.75,
                                    marginBottom: responsiveFieldGap,
                                }}
                            />
                        ) : null}
                        {!isSubmitted ? (
                            <>
                                {!showRail ? (
                                    <div
                                        style={{
                                            display: "flex",
                                            flexDirection: "column",
                                            gap: responsiveFieldGap,
                                            marginBottom:
                                                responsiveSectionGap * 1.25,
                                        }}
                                    >
                                        <div
                                            style={{
                                                ...typography.titleFont,
                                                color: colors.textPrimary,
                                                fontSize:
                                                    responsiveTitleFontSize,
                                                lineHeight: "1.08em",
                                            }}
                                        >
                                            {narrative.productTitle}
                                        </div>
                                        <div
                                            style={{
                                                ...typography.labelFont,
                                                color: colors.textSecondary,
                                                fontSize:
                                                    containerWidth < 480
                                                        ? "14px"
                                                        : undefined,
                                            }}
                                        >
                                            {narrative.supportingLine}
                                        </div>
                                    </div>
                                ) : null}
                                {!showRail && narrative.showStepList ? (
                                    <div
                                        style={{
                                            marginBottom: responsiveSectionGap,
                                        }}
                                    >
                                        {renderProgressBlock(false)}
                                    </div>
                                ) : null}
                                {renderStepBody()}
                            </>
                        ) : (
                            <QuoteSuccessPanel
                                successScreenCopy={successScreenCopy}
                                totalEstimate={totalEstimate}
                                rangeLowPercent={rangeLowPercent}
                                rangeHighPercent={rangeHighPercent}
                                currencySymbol={currencySymbol}
                                isOnCanvas={isOnCanvas}
                                prefersReducedMotion={prefersReducedMotion}
